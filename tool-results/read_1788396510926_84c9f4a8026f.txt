    53→function getLocalizedName(
    54→  item: { name_ar: string; name_fr: string; name_en: string },
    55→  locale: Locale
    56→): string {
    57→  switch (locale) {
    58→    case 'ar': return item.name_ar;

    61→  }
    62→}

    74→// ─── Video Embed Helper ─────────────────────────────────────────
    75→
    76→function getVideoEmbedUrl(url: string): string | null {
    77→  // YouTube: youtube.com/watch?v=ID
    78→  let match = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]{11})/);
    79→  if (match) return `https://www.youtube.com/embed/${match[1]}`;
    80→  // Vimeo: vimeo.com/ID
    81→  match = url.match(/vimeo\.com\/(\d+)/);
    82→  if (match) return `https://player.vimeo.com/video/${match[1]}`;
    83→  return null;
    84→}
    85→
    86→function VideoEmbed({ url }: { url: string }) {
    87→  const embedUrl = getVideoEmbedUrl(url);
    88→  if (!embedUrl) return null;
    89→  return (
    90→    <iframe
    91→      src={embedUrl}
    92→      className="size-full"
    93→      allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
    94→      allowFullScreen
    95→      title="Video preview"
    96→    />
    97→  );
    98→}
