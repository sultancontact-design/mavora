    89→function getLocalizedListingField(
    90→  listing: { title_ar?: string; title_fr?: string; title_en?: string },
    91→  locale: Locale
    92→): string {
    93→  switch (locale) {
    94→    case 'ar': return listing.title_ar || listing.title_en || listing.title_fr || '';
    95→    case 'fr': return listing.title_fr || listing.title_en || listing.title_ar || '';
    96→    default: return listing.title_en || listing.title_ar || listing.title_fr || '';
    97→  }
    98→}
    99→
   100→function getReasonDisplay(reason: string, t: (key: string) => string): string {
   101→  const map: Record<string, string> = {
   102→    scam: t('report.reason_scam_display'),
   103→    prohibited: t('report.reason_prohibited_display'),
   104→    duplicate: t('report.reason_duplicate_display'),
   105→    wrong_category: t('report.reason_wrong_category_display'),
   106→    other: t('report.reason_other_display'),
   107→  };
   108→  return map[reason] ?? reason;
   109→}
   110→
   111→function statusBadgeVariant(status: string): 'default' | 'secondary' | 'outline' | 'destructive' {
   112→  switch (status) {
   113→    case 'active': return 'default';
   114→    case 'draft': return 'secondary';
   115→    case 'archived': return 'outline';
   116→    case 'sold': return 'secondary';
   117→    case 'reserved': return 'outline';
   118→    case 'rejected': return 'destructive';
   119→    case 'pending_review': return 'outline';
   120→    default: return 'secondary';
   121→  }
   122→}

   124→function formatDate(dateStr: string, locale: Locale): string {
   125→  const date = new Date(dateStr);
   126→  return date.toLocaleDateString(locale === 'ar' ? 'ar-MA' : locale === 'fr' ? 'fr-MA' : 'en-US', {
   127→    year: 'numeric',
   128→    month: 'short',
   129→    day: 'numeric',
   130→  });
   131→}
