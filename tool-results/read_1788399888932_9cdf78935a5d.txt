    70→function getLocalizedName(item: { name_ar: string; name_fr: string; name_en: string }, locale: Locale): string {
    71→  switch (locale) {
    72→    case 'ar': return item.name_ar;
    73→    case 'fr': return item.name_fr;
    74→    default: return item.name_en;
    75→  }

    78→function getVideoEmbedUrl(url: string): string | null {
    79→  let match = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]{11})/);
    80→  if (match) return `https://www.youtube.com/embed/${match[1]}`;
    81→  match = url.match(/vimeo\.com\/(\d+)/);
    82→  if (match) return `https://player.vimeo.com/video/${match[1]}`;
    83→  return null;
    84→}

    86→function timeAgo(dateStr: string, locale: Locale): string {
    87→  const now = Date.now();
    88→  const date = new Date(dateStr).getTime();
    89→  const diffMs = now - date;
    90→  const minutes = Math.floor(diffMs / 60000);
    91→  const hours = Math.floor(diffMs / 3600000);
    92→  const days = Math.floor(diffMs / 86400000);
    93→  const months = Math.floor(days / 30);
    94→
    95→  if (locale === 'ar') {
    96→    if (minutes < 1) return 'الآن';
    97→    if (minutes < 60) return `منذ ${minutes} دقيقة`;
    98→    if (hours < 24) return `منذ ${hours} ساعة`;
    99→    if (days < 30) return `منذ ${days} يوم`;
   100→    return `منذ ${months} شهر`;
   101→  }
   102→  if (locale === 'fr') {
   103→    if (minutes < 1) return 'à l\'instant';
   104→    if (minutes < 60) return `il y a ${minutes} min`;
   105→    if (hours < 24) return `il y a ${hours}h`;
   106→    if (days < 30) return `il y a ${days}j`;
   107→    return `il y a ${months}m`;
   108→  }
   109→  if (minutes < 1) return 'just now';
   110→  if (minutes < 60) return `${minutes}m ago`;
   111→  if (hours < 24) return `${hours}h ago`;
   112→  if (days < 30) return `${days}d ago`;
   113→  return `${months}mo ago`;
   114→}
