    70→    70→function getLocalizedName(item: { name_ar: string; name_fr: string; name_en: string }, locale: Locale): string {
    71→    71→  switch (locale) {
    72→    72→    case 'ar': return item.name_ar;
    73→    73→    case 'fr': return item.name_fr;
    74→    74→    default: return item.name_en;
    75→    75→  }
    76→    76→}

    78→    78→function getVideoEmbedUrl(url: string): string | null {
    79→    79→  let match = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]{11})/);
    80→    80→  if (match) return `https://www.youtube.com/embed/${match[1]}`;
    81→    81→  match = url.match(/vimeo\.com\/(\d+)/);
    82→    82→  if (match) return `https://player.vimeo.com/video/${match[1]}`;
    83→    83→  return null;
    84→    84→}

    86→    86→function timeAgo(dateStr: string, locale: Locale): string {
    87→    87→  const now = Date.now();
    88→    88→  const date = new Date(dateStr).getTime();
    89→    89→  const diffMs = now - date;
    90→    90→  const minutes = Math.floor(diffMs / 60000);
    91→    91→  const hours = Math.floor(diffMs / 3600000);
    92→    92→  const days = Math.floor(diffMs / 86400000);
    93→    93→  const months = Math.floor(days / 30);
    94→    94→
    95→    95→  if (locale === 'ar') {
    96→    96→    if (minutes < 1) return 'الآن';
    97→    97→    if (minutes < 60) return `منذ ${minutes} دقيقة`;
    98→    98→    if (hours < 24) return `منذ ${hours} ساعة`;
    99→    99→    if (days < 30) return `منذ ${days} يوم`;
   100→   100→    return `منذ ${months} شهر`;
   101→   101→  }
   102→   102→  if (locale === 'fr') {
   103→   103→    if (minutes < 1) return 'à l\'instant';
   104→   104→    if (minutes < 60) return `il y a ${minutes} min`;
   105→   105→    if (hours < 24) return `il y a ${hours}h`;
   106→   106→    if (days < 30) return `il y a ${days}j`;
   107→   107→    return `il y a ${months}m`;
   108→   108→  }
   109→   109→  if (minutes < 1) return 'just now';
   110→   110→  if (minutes < 60) return `${minutes}m ago`;
   111→   111→  if (hours < 24) return `${hours}h ago`;
   112→   112→  if (days < 30) return `${days}d ago`;
   113→   113→  return `${months}mo ago`;
   114→   114→}

   677→   677→function BrowseView({
   678→   678→  categories,

   680→   680→}: {
   681→   681→  categories: Category[] | null;
   682→   682→  categoriesLoading: boolean;
   683→   683→}) {
   684→   684→  const { t, locale } = useTranslation();
   685→   685→  const { selectedCategoryId, searchQuery, navigateHome, navigateDetail } = useNavigationStore();
   686→   686→  const [listings, setListings] = useState<PaginatedResponse<Listing> | null>(null);
   687→   687→  const [listingsLoading, setListingsLoading] = useState(true);
   688→   688→  const [sortBy, setSortBy] = useState('newest');
   689→   689→  const [page, setPage] = useState(1);
   690→   690→
   691→   691→  // Filter states
   692→   692→  const [countries, setCountries] = useState<Country[] | null>(null);
   693→   693→  const [cities, setCities] = useState<City[] | null>(null);
   694→   694→  const [currencies, setCurrencies] = useState<Currency[] | null>(null);
   695→   695→  const [selectedCountryId, setSelectedCountryId] = useState<string | null>(null);
   696→   696→  const [selectedCityId, setSelectedCityId] = useState<string | null>(null);
   697→   697→  const [minPrice, setMinPrice] = useState('');
   698→   698→  const [maxPrice, setMaxPrice] = useState('');
   699→   699→
   700→   700→  // Fetch countries & currencies on mount
   701→   701→  useEffect(() => {
   702→   702→    Promise.all([
   703→   703→      fetch('/api/countries').then((r) => r.ok ? r.json() : []),
   704→   704→      fetch('/api/currencies').then((r) => r.ok ? r.json() : []),
   705→   705→    ]).then(([countriesData, currenciesData]) => {
   706→   706→      setCountries(countriesData);
   707→   707→      setCurrencies(currenciesData);
   708→   708→    });
   709→   709→  }, []);
   710→   710→
   711→   711→  // Fetch cities when country changes
   712→   712→  useEffect(() => {
   713→   713→    if (!selectedCountryId) {
   714→   714→      setCities(null);
   715→   715→      return;
   716→   716→    }
   717→   717→    fetch(`/api/cities?country_id=${selectedCountryId}`)
   718→   718→      .then((r) => r.ok ? r.json() : [])
   719→   719→      .then((data) => setCities(data));
   720→   720→  }, [selectedCountryId]);
   721→   721→
   722→   722→  // Compute active filter count
   723→   723→  const activeFilterCount = [
   724→   724→    selectedCountryId,
   725→   725→    selectedCityId,
   726→   726→    minPrice,
   727→   727→    maxPrice,
   728→   728→  ].filter(Boolean).length;
   729→   729→
   730→   730→  const fetchListings = useCallback(async () => {
   731→   731→    setListingsLoading(true);
   732→   732→    try {
   733→   733→      const params = new URLSearchParams({
   734→   734→        sort_by: sortBy,
   735→   735→        page: String(page),
   736→   736→        per_page: '12',
   737→   737→      });
   738→   738→      if (selectedCategoryId) params.set('category_id', selectedCategoryId);
   739→   739→      if (searchQuery) params.set('search', searchQuery);

   741→   741→      if (selectedCityId) params.set('city_id', selectedCityId);
   742→   742→      if (minPrice) params.set('min_price', minPrice);
   743→   743→      if (maxPrice) params.set('max_price', maxPrice);

   745→   745→      const res = await fetch(`/api/listings?${params}`);

   747→   747→        const data = await res.json();
   748→   748→        setListings(data);
   749→   749→      }
   750→   750→    } catch {
   751→   751→      // silent
   752→   752→    } finally {
   753→   753→      setListingsLoading(false);
   754→   754→    }
   755→   755→  }, [selectedCategoryId, searchQuery, sortBy, page, selectedCountryId, selectedCityId, minPrice, maxPrice]);
