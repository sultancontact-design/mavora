    82→function getLocalizedName(item: { name_ar: string; name_fr: string; name_en: string }, locale: Locale): string {
    83→  switch (locale) {

    86→    default: return item.name_en;
    87→  }

    90→function getVideoEmbedUrl(url: string): string | null {
    91→  let match = url.match(/(?:youtube\.com\/watch\?v=|youtu\.be\/)([a-zA-Z0-9_-]{11})/);
    92→  if (match) return `https://www.youtube.com/embed/${match[1]}`;
    93→  match = url.match(/vimeo\.com\/(\d+)/);
    94→  if (match) return `https://player.vimeo.com/video/${match[1]}`;
    95→  return null;
    96→}

    98→function timeAgo(dateStr: string, locale: Locale): string {
    99→  const now = Date.now();
   100→  const date = new Date(dateStr).getTime();
   101→  const diffMs = now - date;
   102→  const minutes = Math.floor(diffMs / 60000);
   103→  const hours = Math.floor(diffMs / 3600000);
   104→  const days = Math.floor(diffMs / 86400000);
   105→  const months = Math.floor(days / 30);
   106→
   107→  if (locale === 'ar') {
   108→    if (minutes < 1) return 'الآن';
   109→    if (minutes < 60) return `منذ ${minutes} دقيقة`;
   110→    if (hours < 24) return `منذ ${hours} ساعة`;
   111→    if (days < 30) return `منذ ${days} يوم`;
   112→    return `منذ ${months} شهر`;
   113→  }
   114→  if (locale === 'fr') {
   115→    if (minutes < 1) return 'à l\'instant';
   116→    if (minutes < 60) return `il y a ${minutes} min`;
   117→    if (hours < 24) return `il y a ${hours}h`;
   118→    if (days < 30) return `il y a ${days}j`;
   119→    return `il y a ${months}m`;
   120→  }
   121→  if (minutes < 1) return 'just now';
   122→  if (minutes < 60) return `${minutes}m ago`;
   123→  if (hours < 24) return `${hours}h ago`;
   124→  if (days < 30) return `${days}d ago`;
   125→  return `${months}mo ago`;
   126→}

  1140→function BrowseView({
  1141→  categories,
  1142→  categoriesLoading,
  1143→}: {
  1144→  categories: Category[] | null;
  1145→  categoriesLoading: boolean;
  1146→}) {
  1147→  const { t, locale } = useTranslation();
  1148→  const { selectedCategoryId, searchQuery, navigateHome, navigateDetail } = useNavigationStore();
  1149→  const [listings, setListings] = useState<PaginatedResponse<Listing> | null>(null);
  1150→  const [listingsLoading, setListingsLoading] = useState(true);
  1151→  const [sortBy, setSortBy] = useState('newest');
  1152→  const [page, setPage] = useState(1);
  1153→
  1154→  // Filter states
  1155→  const [countries, setCountries] = useState<Country[] | null>(null);
  1156→  const [cities, setCities] = useState<City[] | null>(null);
  1157→  const [currencies, setCurrencies] = useState<Currency[] | null>(null);
  1158→  const [selectedCountryId, setSelectedCountryId] = useState<string | null>(null);
  1159→  const [selectedCityId, setSelectedCityId] = useState<string | null>(null);
  1160→  const [minPrice, setMinPrice] = useState('');
  1161→  const [maxPrice, setMaxPrice] = useState('');
  1162→
  1163→  // Fetch countries & currencies on mount
  1164→  useEffect(() => {
  1165→    Promise.all([
  1166→      fetch('/api/countries').then((r) => r.ok ? r.json() : []),
  1167→      fetch('/api/currencies').then((r) => r.ok ? r.json() : []),
  1168→    ]).then(([countriesData, currenciesData]) => {
  1169→      setCountries(countriesData);
  1170→      setCurrencies(currenciesData);
  1171→    });
  1172→  }, []);
  1173→
  1174→  // Fetch cities when country changes
  1175→  useEffect(() => {
  1176→    if (!selectedCountryId) {
  1177→      setCities(null);
  1178→      return;
  1179→    }
  1180→    fetch(`/api/cities?country_id=${selectedCountryId}`)
  1181→      .then((r) => r.ok ? r.json() : [])
  1182→      .then((data) => setCities(data));
  1183→  }, [selectedCountryId]);
  1184→
  1185→  // Compute active filter count
  1186→  const activeFilterCount = [
  1187→    selectedCountryId,
  1188→    selectedCityId,
  1189→    minPrice,
  1190→    maxPrice,
  1191→  ].filter(Boolean).length;
  1192→
  1193→  const fetchListings = useCallback(async () => {
  1194→    setListingsLoading(true);
  1195→    try {
  1196→      const params = new URLSearchParams({
  1197→        sort_by: sortBy,
  1198→        page: String(page),
  1199→        per_page: '12',
  1200→      });
  1201→      if (selectedCategoryId) params.set('category_id', selectedCategoryId);
  1202→      if (searchQuery) params.set('search', searchQuery);
  1203→      if (selectedCountryId) params.set('country_id', selectedCountryId);
  1204→      if (selectedCityId) params.set('city_id', selectedCityId);
  1205→      if (minPrice) params.set('min_price', minPrice);
  1206→      if (maxPrice) params.set('max_price', maxPrice);
  1207→
  1208→      const res = await fetch(`/api/listings?${params}`);
  1209→      if (res.ok) {
  1210→        const data = await res.json();
  1211→        setListings(data);
  1212→      }
  1213→    } catch {
  1214→      // silent
  1215→    } finally {
  1216→      setListingsLoading(false);
  1217→    }
  1218→  }, [selectedCategoryId, searchQuery, sortBy, page, selectedCountryId, selectedCityId, minPrice, maxPrice]);
  1219→
  1220→  useEffect(() => {
  1221→    fetchListings();
  1222→  }, [fetchListings]);
  1223→
  1224→  // Reset page on filter change
  1225→  useEffect(() => {
  1226→    setPage(1);
  1227→  }, [selectedCategoryId, searchQuery, sortBy, selectedCountryId, selectedCityId, minPrice, maxPrice]);
  1228→
  1229→  const handleClearAllFilters = useCallback(() => {
  1230→    setSelectedCountryId(null);
  1231→    setSelectedCityId(null);
  1232→    setMinPrice('');
  1233→    setMaxPrice('');
  1234→  }, []);
  1235→
  1236→  const selectedCategory = categories?.find((c) => c.id === selectedCategoryId);
  1237→
  1238→  return (
  1239→    <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
  1240→      {/* Back + Title */}
  1241→      <div className="mb-6 flex items-center gap-3">
  1242→        <Button
  1243→          variant="ghost"
  1244→          size="sm"
  1245→          onClick={navigateHome}
  1246→          className="gap-1.5 text-muted-foreground hover:text-foreground"
  1247→        >
  1248→          <ArrowLeft className="size-4" />
  1249→          {t('common.back')}
  1250→        </Button>
  1251→      </div>
  1252→
  1253→      <div className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
  1254→        <div className="flex items-center gap-2">
  1255→          <h1 className="text-xl font-bold text-foreground sm:text-2xl">
  1256→            {searchQuery
  1257→              ? `${t('listings.search_results')} "${searchQuery}"`
  1258→              : selectedCategory
  1259→                ? getLocalizedName(selectedCategory, locale)
  1260→                : t('listings.latest')}
  1261→          </h1>
  1262→          {listings && (
  1263→            <Badge variant="secondary" className="text-xs">

  1269→        {/* Sort */}

  1273→            onChange={(e) => setSortBy(e.target.value)}
