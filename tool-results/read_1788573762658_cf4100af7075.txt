   141→// Wrapper for lazy components with error handling
   142→function LazyTab({ 
   143→  component: Component, 
   144→  onError 
   145→}: { 
   146→  component: ComponentType; 
   147→  onError: () => void;
   148→}) {
   149→  const [hasError, setHasError] = useState(false);
   150→
   151→  if (hasError) {
   152→    return <ErrorFallback onRetry={() => setHasError(false)} />;
   153→  }
   154→
   155→  return (
   156→    <Suspense fallback={<TabLoader />}>
   157→      <ErrorCatcher onError={() => setHasError(true)}>
   158→        <Component />
   159→      </ErrorCatcher>
   160→    </Suspense>
   161→  );
   162→}
   163→
   164→// Error catcher component
   165→class ErrorCatcher extends React.Component<
   166→  { children: React.ReactNode; onError: () => void },
   167→  { hasError: boolean }
   168→> {
   169→  constructor(props: { children: React.ReactNode; onError: () => void }) {
   170→    super(props);
   171→    this.state = { hasError: false };
   172→  }
   173→
   174→  static getDerivedStateFromError() {
   175→    return { hasError: true };
   176→  }
   177→
   178→  componentDidCatch() {
   179→    this.props.onError();
   180→  }
   181→
   182→  render() {
   183→    if (this.state.hasError) return null;
   184→    return this.props.children;
   185→  }
   186→}
