    98→function StatCard({ 
    99→  title, 
   100→  value, 
   101→  icon: Icon, 
   102→  trend, 
   103→  trendValue, 
   104→  color = 'blue' 
   105→}: { 
   106→  title: string; 
   107→  value: string | number; 
   108→  icon: any; 
   109→  trend?: 'up' | 'down';
   110→  trendValue?: string;
   111→  color?: string;
   112→}) {
   113→  const colors: Record<string, { bg: string; icon: string; text: string }> = {
   114→    blue: { bg: 'bg-blue-50 dark:bg-blue-900/20', icon: 'text-blue-600', text: 'text-blue-700 dark:text-blue-300' },
   115→    green: { bg: 'bg-green-50 dark:bg-green-900/20', icon: 'text-green-600', text: 'text-green-700 dark:text-green-300' },
   116→    purple: { bg: 'bg-purple-50 dark:bg-purple-900/20', icon: 'text-purple-600', text: 'text-purple-700 dark:text-purple-300' },
   117→    orange: { bg: 'bg-orange-50 dark:bg-orange-900/20', icon: 'text-orange-600', text: 'text-orange-700 dark:text-orange-300' },
   118→    red: { bg: 'bg-red-50 dark:bg-red-900/20', icon: 'text-red-600', text: 'text-red-700 dark:text-red-300' },
   119→  };
   120→
   121→  const c = colors[color] || colors.blue;
   122→
   123→  return (
   124→    <div className="bg-white dark:bg-gray-800 rounded-xl shadow-sm border border-gray-200 dark:border-gray-700 p-6 hover:shadow-md transition-shadow">
   125→      <div className="flex items-center justify-between">
   126→        <div>
   127→          <p className="text-sm font-medium text-gray-500 dark:text-gray-400">{title}</p>
   128→          <p className={`text-2xl font-bold mt-1 ${c.text}`}>{value}</p>
   129→          {trend && trendValue && (
   130→            <div className={`flex items-center mt-2 text-sm ${trend === 'up' ? 'text-green-600' : 'text-red-600'}`}>
   131→              {trend === 'up' ? <ArrowUpRight className="w-4 h-4" /> : <ArrowDownRight className="w-4 h-4" />}
   132→              <span className="ms-1">{trendValue}</span>
   133→            </div>
   134→          )}
   135→        </div>
   136→        <div className={`p-3 rounded-xl ${c.bg}`}>
   137→          <Icon className={`w-6 h-6 ${c.icon}`} />
   138→        </div>
   139→      </div>
   140→    </div>
   141→  );
   142→}
