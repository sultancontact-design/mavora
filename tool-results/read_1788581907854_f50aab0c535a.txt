   301→  // Data states

   305→  const [stats, setStats] = useState<DashboardStats>(DEFAULT_STATS);
   306→  
   307→  // UI states
   308→  const [isLoading, setIsLoading] = useState(true);
   309→  const [isRefreshing, setIsRefreshing] = useState(false);
   310→  const [error, setError] = useState<string | null>(null);
   311→  const [usingMockData, setUsingMockData] = useState(false);
   312→
   313→  // ============================================================
   314→  // Data Fetching Functions
   315→  // ============================================================
   316→
   317→  const fetchDashboardData = useCallback(async (showRefreshIndicator = false) => {
   318→    if (showRefreshIndicator) {
   319→      setIsRefreshing(true);
   320→    } else {
   321→      setIsLoading(true);
   322→    }
   323→    setError(null);
   324→
   325→    try {
   326→      // Fetch stats from API
   327→      const statsResponse = await fetch('/api/admin/stats');
   328→      if (statsResponse.ok) {
   329→        const statsData = await statsResponse.json();
   330→        setStats({
   331→          totalUsers: statsData.totalUsers || DEFAULT_STATS.totalUsers,
   332→          activeListings: statsData.activeListings || DEFAULT_STATS.activeListings,
   333→          totalRevenue: statsData.totalRevenue || DEFAULT_STATS.totalRevenue,
   334→          pendingOrders: statsData.pendingOrders || DEFAULT_STATS.pendingOrders,
   335→          monthlyGrowth: statsData.monthlyGrowth || DEFAULT_STATS.monthlyGrowth,
   336→        });
   337→        setUsingMockData(false);
   338→      } else {
   339→        console.warn('[Dashboard] Stats API returned non-OK, using defaults');
   340→        setStats(DEFAULT_STATS);
   341→        setUsingMockData(true);
   342→      }
   343→
   344→      // Fetch users from API
   345→      try {
   346→        const usersResponse = await fetch('/api/admin/users');
   347→        if (usersResponse.ok) {
   348→          const usersData = await usersResponse.json();
   349→          if (Array.isArray(usersData.users) && usersData.users.length > 0) {
   350→            setUsers(usersData.users.map((u: Record<string, unknown>) => ({
   351→              id: u.id as string,
   352→              name: (u.name || u.display_name || 'Unknown') as string,
   353→              email: u.email as string,
   354→              role: (u.role || 'user') as User['role'],
   355→              status: (u.isActive !== false ? 'active' : 'suspended') as User['status'],
   356→              joinDate: u.createdAt ? new Date(u.createdAt as string).toLocaleDateString('ar-MA') : '-',
   357→              lastLogin: u.lastLoginAt ? new Date(u.lastLoginAt as string).toLocaleDateString('ar-MA') : '-',
   358→              listingsCount: u.listingsCount || u._count?.listings || 0,
   359→            })));
   360→          } else {
   361→            setUsers(MOCK_USERS);
   362→            setUsingMockData(true);
   363→          }
   364→        } else {
   365→          setUsers(MOCK_USERS);
   366→          setUsingMockData(true);
   367→        }
   368→      } catch (usersError) {
   369→        console.warn('[Dashboard] Users API failed, using mock data:', usersError);
   370→        setUsers(MOCK_USERS);
   371→        setUsingMockData(true);
   372→      }
   373→
   374→      // Fetch listings from API
   375→      try {
   376→        const listingsResponse = await fetch('/api/listings?limit=10');
   377→        if (listingsResponse.ok) {
   378→          const listingsData = await listingsResponse.json();
   379→          if (Array.isArray(listingsData.listings) && listingsData.listings.length > 0) {
   380→            setListings(listingsData.listings.map((l: Record<string, unknown>) => ({
   381→              id: l.id as string,
   382→              title: l.title as string,
   383→              category: l.category?.name || l.categoryId || 'غير مصنف',
   384→              price: Number(l.price) || 0,
   385→              seller: l.seller?.display_name || 'غير معروف',
   386→              status: (l.status || 'active') as Listing['status'],
   387→              views: l.viewCount || 0,
   388→              createdAt: l.createdAt ? new Date(l.createdAt as string).toLocaleDateString('ar-MA') : '-',
   389→            })));
   390→          } else {
   391→            setListings(MOCK_LISTINGS);
   392→            setUsingMockData(true);
   393→          }
   394→        } else {
   395→          setListings(MOCK_LISTINGS);
   396→          setUsingMockData(true);
   397→        }
   398→      } catch (listingsError) {
   399→        console.warn('[Dashboard] Listings API failed, using mock data:', listingsError);
   400→        setListings(MOCK_LISTINGS);
   401→        setUsingMockData(true);
   402→      }
   403→
   404→      // Orders - use mock for now (API may not be fully implemented)
   405→      setOrders(MOCK_ORDERS);
   406→
   407→    } catch (err) {
   408→      console.error('[Dashboard] Failed to fetch data:', err);
   409→      setError('فشل في تحميل البيانات. يرجى التحقق من اتصال الإنترنت.');
   410→      // Fall back to mock data on any error
   411→      setUsers(MOCK_USERS);
   412→      setListings(MOCK_LISTINGS);
   413→      setOrders(MOCK_ORDERS);
   414→      setStats(DEFAULT_STATS);
   415→      setUsingMockData(true);
   416→    } finally {
   417→      setIsLoading(false);
   418→      setIsRefreshing(false);
   419→    }
   420→  }, []);
   421→
   422→  // Initial data fetch
   423→  useEffect(() => {
   424→    fetchDashboardData();
   425→  }, [fetchDashboardData]);
   426→
   427→  // Auto-refresh every 60 seconds
   428→  useEffect(() => {
   429→    const interval = setInterval(() => {
   430→      fetchDashboardData(true);
   431→    }, 60000);
   432→    return () => clearInterval(interval);
   433→  }, [fetchDashboardData]);
   434→
   435→  // Handle logout
   436→  const handleLogout = () => {
   437→    if (typeof window !== 'undefined') {
   438→      localStorage.removeItem('mavora_user');
   439→      localStorage.removeItem('mavora_auth_token');
   440→      document.cookie = 'mavora_admin_session=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
   441→      document.cookie = 'mavora_demo_mode=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
   442→    }
   443→    router.push('/admin-login');
   444→  };
   445→
   446→  // Handle refresh
   447→  const handleRefresh = () => {
   448→    fetchDashboardData(true);
   449→  };
   450→
   451→  // Navigate to listing detail
   452→  const handleListingClick = (listingId: string) => {
   453→    router.push(`/listings/${listingId}`);
   454→  };
   455→
   456→  // Navigate to user profile
   457→  const handleUserClick = (userId: string) => {
   458→    window.open(`/profile?id=${userId}`, '_blank');
   459→  };
   460→
