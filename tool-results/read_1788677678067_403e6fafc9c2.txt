   137→function StatCard({ 
   138→  title, 
   139→  value, 
   140→  icon: Icon, 
   141→  trend, 
   142→  trendValue, 
   143→  color = 'blue',
   144→  isLoading = false

   146→  title: string; 
   147→  value: string | number; 
   148→  icon: any; 
   149→  trend?: 'up' | 'down';

   151→  color?: string;

   153→}) {
   154→  const colors: Record<string, { bg: string; icon: string; text: string }> = {
   155→    blue: { bg: 'bg-blue-50 dark:bg-blue-900/20', icon: 'text-blue-600', text: 'text-blue-700 dark:text-blue-300' },
   156→    green: { bg: 'bg-green-50 dark:bg-green-900/20', icon: 'text-green-600', text: 'text-green-700 dark:text-green-300' },

   176→  }
   177→
   178→  return (
   179→    <div className="bg-white dark:bg-gray-800 rounded-xl shadow-sm border border-gray-200 dark:border-gray-700 p-6 hover:shadow-md transition-shadow">

   234→  badge,
   235→  isLoading = false
   236→}: {

   251→          : 'text-gray-600 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-gray-700/50'
   252→      } ${isLoading ? 'opacity-50 cursor-wait' : ''}`}
   253→    >
   254→      {isLoading ? (
   255→        <Loader2 className="w-5 h-5 animate-spin" />
   256→      ) : (
   257→        <Icon className="w-5 h-5" />
   258→      )}
   259→      <span className="font-medium">{label}</span>
   260→      {badge !== undefined && !isLoading && (
   261→        <span className={`ms-auto px-2 py-0.5 rounded-full text-xs font-medium ${
   262→          active ? 'bg-white/20 text-white' : 'bg-gray-200 dark:bg-gray-600 text-gray-600 dark:text-gray-300'
   263→        }`}>
   264→          {badge}
   265→        </span>
   266→      )}
   267→    </button>
   268→  );
   269→}

   320→export default function SuperAdminDashboard() {
   321→  const router = useRouter();
   322→  const [activeTab, setActiveTab] = useState('dashboard');
   323→  const [searchQuery, setSearchQuery] = useState('');
   324→  
   325→  // Data states

   328→  const [orders, setOrders] = useState<Order[]>([]);
   329→  const [stats, setStats] = useState<DashboardStats>(DEFAULT_STATS);
   330→  
   331→  // UI states
   332→  const [isLoading, setIsLoading] = useState(true);
   333→  const [isRefreshing, setIsRefreshing] = useState(false);
   334→  const [error, setError] = useState<string | null>(null);
   335→  const [usingMockData, setUsingMockData] = useState(false);
   336→
   337→  // ============================================================
   338→  // Data Fetching Functions
   339→  // ============================================================
   340→
   341→  const fetchDashboardData = useCallback(async (showRefreshIndicator = false) => {
   342→    if (showRefreshIndicator) {
   343→      setIsRefreshing(true);
   344→    } else {
   345→      setIsLoading(true);
   346→    }
   347→    setError(null);
   348→
   349→    try {
   350→      // Try to fetch stats from API with timeout
   351→      let statsFetched = false;
   352→      try {
   353→        const controller = new AbortController();
   354→        const timeoutId = setTimeout(() => controller.abort(), 8000); // 8s timeout
   355→        
   356→        const statsResponse = await fetch('/api/admin/stats', { 
   357→          signal: controller.signal
   358→        });
   359→        clearTimeout(timeoutId);
   360→        
   361→        if (statsResponse.ok) {
   362→          const statsData = await statsResponse.json();
   363→          if (statsData.success && statsData.overview) {
   364→            setStats({
   365→              totalUsers: statsData.overview.total_users || DEFAULT_STATS.totalUsers,
   366→              activeListings: statsData.overview.active_listings || DEFAULT_STATS.activeListings,
   367→              totalRevenue: statsData.overview.total_revenue || DEFAULT_STATS.totalRevenue,
   368→              pendingOrders: statsData.overview.pending_reports || DEFAULT_STATS.pendingOrders,
   369→              monthlyGrowth: 15.5, // Would calculate from real data
   370→            });
   371→            statsFetched = true;
   372→            setUsingMockData(false);
   373→          }
   374→        }
   375→      } catch (statsError) {
   376→        console.warn('[Dashboard] Stats API failed (using defaults):', statsError instanceof Error ? statsError.message : statsError);
   377→      }
   378→      
   379→      if (!statsFetched) {
   380→        setStats(DEFAULT_STATS);
   381→        setUsingMockData(true);
   382→      }
   383→
   384→      // Try to fetch users from API
   385→      let usersFetched = false;
   386→      try {
   387→        const controller = new AbortController();
   388→        const timeoutId = setTimeout(() => controller.abort(), 5000);
   389→        
   390→        const usersResponse = await fetch('/api/admin/users', { 
   391→          signal: controller.signal 
   392→        });
   393→        clearTimeout(timeoutId);
   394→        
   395→        if (usersResponse.ok) {
   396→          const usersData = await usersResponse.json();
   397→          const usersArray = usersData.users || usersData;
   398→          if (Array.isArray(usersArray) && usersArray.length > 0) {
   399→            setUsers(usersArray.map((u: Record<string, unknown>) => ({
   400→              id: u.id as string,
   401→              name: (u.name || u.display_name || 'Unknown') as string,
   402→              email: u.email as string,
   403→              role: (u.role || 'user') as User['role'],
   404→              status: (u.isActive !== false ? 'active' : 'suspended') as User['status'],
   405→              joinDate: u.createdAt ? new Date(u.createdAt as string).toLocaleDateString('ar-MA') : '-',
   406→              lastLogin: u.lastLoginAt ? new Date(u.lastLoginAt as string).toLocaleDateString('ar-MA') : '-',
   407→              listingsCount: u.listingsCount || u._count?.listings || 0,
   408→            })));
   409→            usersFetched = true;
   410→          }
   411→        }
   412→      } catch (usersError) {
   413→        // Silent fail - will use mock
   414→      }
   415→      
   416→      if (!usersFetched) {
   417→        setUsers(MOCK_USERS);
   418→        setUsingMockData(true);
   419→      }
   420→
   421→      // Try to fetch listings from API
   422→      let listingsFetched = false;
   423→      try {
   424→        const controller = new AbortController();
   425→        const timeoutId = setTimeout(() => controller.abort(), 5000);
   426→        
   427→        const listingsResponse = await fetch('/api/listings?limit=10', { 
   428→          signal: controller.signal 
   429→        });
   430→        clearTimeout(timeoutId);
   431→        
   432→        if (listingsResponse.ok) {
   433→          const listingsData = await listingsResponse.json();
   434→          if (Array.isArray(listingsData.listings) && listingsData.listings.length > 0) {
   435→            setListings(listingsData.listings.map((l: Record<string, unknown>) => ({
   436→              id: l.id as string,
   437→              title: l.title as string,
   438→              category: l.category?.name || l.categoryId || 'غير مصنف',
   439→              price: Number(l.price) || 0,
   440→              seller: l.seller?.display_name || 'غير معروف',
   441→              status: (l.status || 'active') as Listing['status'],
   442→              views: l.viewCount || 0,
   443→              createdAt: l.createdAt ? new Date(l.createdAt as string).toLocaleDateString('ar-MA') : '-',
   444→              image: getCategoryEmoji(l.category?.name || l.categoryId || 'غير مصنف'),
   445→            })));
   446→            listingsFetched = true;
   447→          }
   448→        }
   449→      } catch (listingsError) {
   450→        // Silent fail - will use mock
   451→      }
   452→      
   453→      if (!listingsFetched) {
   454→        setListings(MOCK_LISTINGS);
   455→        setUsingMockData(true);
   456→      }
   457→
   458→      // Orders - use mock for now (orders API may not be fully implemented)
   459→      setOrders(MOCK_ORDERS);
   460→
   461→    } catch (err) {
