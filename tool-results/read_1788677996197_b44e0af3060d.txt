    56→    56→function getLocalizedName(
    57→    57→  item: { name_ar?: string; name_fr?: string; name_en?: string },
    58→    58→  locale: Locale
    59→    59→): string {
    60→    60→  switch (locale) {
    61→    61→    case 'ar': return item.name_ar ?? item.name_en ?? '';
    62→    62→    case 'fr': return item.name_fr ?? item.name_en ?? '';
    63→    63→    default: return item.name_en ?? '';
    64→    64→  }
    65→    65→}

    67→    67→function formatPrice(
    68→    68→  price: number | null | undefined,
    69→    69→  currency: Currency | null | undefined,
    70→    70→  locale: Locale
    71→    71→): string {
    72→    72→  if (price === null || price === undefined) {
    73→    73→    return locale === 'ar' ? 'مجاني' : locale === 'fr' ? 'Gratuit' : 'Free';
    74→    74→  }
    75→    75→  
    76→    76→  const symbol = currency?.symbol ?? '';
    77→    77→  const formatted = price.toLocaleString(locale);
    78→    78→  
    79→    79→  return locale === 'ar' ? `${formatted} ${symbol}` : `${symbol}${formatted}`;
    80→    80→}
    81→    81→
    82→    82→function formatDate(dateString: string, locale: Locale): string {
    83→    83→  const date = new Date(dateString);
    84→    84→  return date.toLocaleDateString(
    85→    85→    locale === 'ar' ? 'ar-MA' : locale === 'fr' ? 'fr-FR' : 'en-US',
    86→    86→    { year: 'numeric', month: 'long', day: 'numeric' }
    87→    87→  );
    88→    88→}

   280→   280→export default function ListingDetail({ listingId: propListingId }: ListingDetailProps) {
   281→   281→  const params = useParams();

   283→   283→  const { t, locale } = useTranslation();
   284→   284→  const { user, isAuthenticated } = useAuthStore();
   285→   285→
   286→   286→  const listingId = propListingId || (params?.id as string);
   287→   287→  
   288→   288→  const [listing, setListing] = useState<ExtendedListing | null>(null);
   289→   289→  const [isLoading, setIsLoading] = useState(true);
   290→   290→  const [error, setError] = useState<string | null>(null);
   291→   291→  const [isFavorited, setIsFavorited] = useState(false);
   292→   292→  const [isFavoriteLoading, setIsFavoriteLoading] = useState(false);
   293→   293→  const [showReportDialog, setShowReportDialog] = useState(false);
   294→   294→
   295→   295→  // Fetch listing data
   296→   296→  const fetchListing = useCallback(async () => {
   297→   297→    if (!listingId) return;
   298→   298→
   299→   299→    setIsLoading(true);
   300→   300→    setError(null);
   301→   301→
   302→   302→    try {
   303→   303→      const res = await fetch(`/api/listings/${listingId}`);
   304→   304→      const data = await res.json();
   305→   305→
   306→   306→      if (!res.ok) {
   307→   307→        throw new Error(data.error || 'Failed to fetch listing');
   308→   308→      }
   309→   309→
   310→   310→      setListing(data);
   311→   311→    } catch (err) {
   312→   312→      setError(err instanceof Error ? err.message : 'An error occurred');
   313→   313→    } finally {
   314→   314→      setIsLoading(false);
   315→   315→    }
   316→   316→  }, [listingId]);
   317→   317→
   318→   318→  // Check favorite status
   319→   319→  const checkFavoriteStatus = useCallback(async () => {
   320→   320→    if (!listingId || !isAuthenticated || !user) return;
   321→   321→
   322→   322→    try {
   323→   323→      const res = await fetch(`/api/favorites?listing_id=${listingId}`);
   324→   324→      if (res.ok) {
   325→   325→        const data = await res.json();
   326→   326→        // Check if this listing is in favorites
   327→   327→        const isFav = Array.isArray(data) && data.some((f: { listing_id: string }) => f.listing_id === listingId);
   328→   328→        setIsFavorited(isFav);
   329→   329→      }
   330→   330→    } catch {
   331→   331→      // Ignore errors for favorite check
   332→   332→    }
   333→   333→  }, [listingId, isAuthenticated, user]);
   334→   334→
   335→   335→  useEffect(() => {
   336→   336→    fetchListing();
   337→   337→    checkFavoriteStatus();
   338→   338→  }, [fetchListing, checkFavoriteStatus]);
   339→   339→
   340→   340→  // Handle favorite toggle
   341→   341→  const handleFavoriteToggle = async () => {
   342→   342→    if (!isAuthenticated || !user || !listingId) {
   343→   343→      toast.error(t('favorites.login_required'));
   344→   344→      return;
   345→   345→    }
   346→   346→
   347→   347→    setIsFavoriteLoading(true);
   348→   348→    try {
   349→   349→      const res = await fetch(`/api/listings/${listingId}/favorite`, {
   350→   350→        method: 'POST',
   351→   351→      });
   352→   352→
   353→   353→      if (!res.ok) throw new Error('Failed to update favorite');
   354→   354→
   355→   355→      const data = await res.json();
   356→   356→      setIsFavorited(data.favorited);
   357→   357→      toast.success(data.favorited ? t('favorites.added') : t('favorites.removed'));
   358→   358→    } catch {
   359→   359→      toast.error(t('common.error'));
   360→   360→    } finally {
   361→   361→      setIsFavoriteLoading(false);
   362→   362→    }
   363→   363→  };
   364→   364→
   365→   365→  // Handle share
   366→   366→  const handleShare = async () => {
   367→   367→    if (navigator.share) {
   368→   368→      try {
   369→   369→        await navigator.share({
   370→   370→          title: listing?.title,
   371→   371→          text: listing?.description?.slice(0, 150),
   372→   372→          url: window.location.href,
   373→   373→        });
   374→   374→      } catch {
   375→   375→        // User cancelled or error
   376→   376→      }
   377→   377→    } else {
   378→   378→      // Fallback: copy to clipboard
   379→   379→      navigator.clipboard.writeText(window.location.href);
   380→   380→      toast.success('Link copied to clipboard');
   381→   381→    }
   382→   382→  };
   383→   383→
   384→   384→  // Handle contact seller

   388→   388→      return;
   389→   389→    }
   390→   390→    // Navigate to messages or open conversation

   392→   392→  };
   393→   393→
   394→   394→  // Loading state
   395→   395→  if (isLoading) {

   397→   397→      <div className="mx-auto max-w-6xl px-4 py-8 sm:px-6 lg:px-8">
   398→   398→        <div className="grid gap-8 lg:grid-cols-5">
   399→   399→          <div className="lg:col-span-3 space-y-4">
   400→   400→            <Skeleton className="aspect-[4/3] w-full rounded-xl" />
   401→   401→            <Skeleton className="h-8 w-3/4" />

   403→   403→            <Skeleton className="h-4 w-5/6" />
   404→   404→          </div>

   406→   406→            <Skeleton className="h-12 w-full rounded-xl" />
   407→   407→            <Skeleton className="h-32 w-full rounded-xl" />
   408→   408→            <Skeleton className="h-48 w-full rounded-xl" />
   409→   409→          </div>

   428→   428→
   429→   429→  // Get localized values
   430→   430→  const categoryName = getLocalizedName(listing.category ?? {}, locale);
   431→   431→  const sellerName = listing.seller?.display_name ?? 'Unknown Seller';
   432→   432→
   433→   433→  return (
   434→   434→    <motion.div
   435→   435→      className="mx-auto max-w-6xl px-4 py-8 sm:px-6 lg:px-8"
   436→   436→      initial="initial"
   437→   437→      animate="animate"
   438→   438→      exit="exit"
   439→   439→      variants={fadeIn}
   440→   440→    >
   441→   441→      {/* Breadcrumb / Back */}

   447→   447→        {t('listings.back_to_browse')}
   448→   448→      </button>
   449→   449→
   450→   450→      <div className="grid gap-8 lg:grid-cols-5">
   451→   451→        {/* Left Column - Images & Description */}
