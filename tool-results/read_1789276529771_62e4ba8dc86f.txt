   143→function StatCard({ 
   144→  title, 
   145→  value, 
   146→  icon: Icon, 
   147→  change, 
   148→  changeType,
   149→  suffix = '',
   150→  prefix = ''
   151→}: { 
   152→  title: string; 
   153→  value: string | number; 
   154→  icon: any; 
   155→  change?: number; 
   156→  changeType?: 'positive' | 'negative' | 'neutral';
   157→  suffix?: string;
   158→  prefix?: string;
   159→}) {
   160→  return (
   161→    <Card>
   162→      <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
   163→        <CardTitle className="text-sm font-medium">{title}</CardTitle>
   164→        <Icon className="h-4 w-4 text-muted-foreground" />
   165→      </CardHeader>
   166→      <CardContent>
   167→        <div className="text-2xl font-bold">{prefix}{value}{suffix}</div>
   168→        {change !== undefined && (
   169→          <p className={`text-xs ${changeType === 'positive' ? 'text-green-600' : changeType === 'negative' ? 'text-red-600' : 'text-gray-500'}`}>
   170→            {changeType === 'positive' ? '+' : ''}{change}% من الأسبوع الماضي
   171→          </p>
   172→        )}
   173→      </CardContent>
   174→    </Card>
   175→  );
   176→}
   177→
   178→function StatusBadge({ status }: { status: string }) {
   179→  const variants: Record<string, { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' }> = {
   180→    active: { label: 'نشط', variant: 'default' },
   181→    pending: { label: 'قيد المراجعة', variant: 'secondary' },
   182→    sold: { label: 'مباع', variant: 'outline' },
   183→    expired: { label: 'منتهي', variant: 'secondary' },
   184→    rejected: { label: 'مرفوض', variant: 'destructive' },
   185→    confirmed: { label: 'مؤكد', variant: 'default' },
   186→    shipped: { label: 'تم الشحن', variant: 'secondary' },
   187→    delivered: { label: 'تم التوصيل', variant: 'outline' },
   188→    cancelled: { label: 'ملغي', variant: 'destructive' },
   189→    pending: { label: 'قيد الانتظار', variant: 'secondary' },
   190→  };
   191→
   192→  const config = variants[status] || { label: status, variant: 'secondary' as const };
   193→  
   194→  return <Badge variant={config.variant}>{config.label}</Badge>;
   195→}
